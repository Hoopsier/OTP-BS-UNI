CREATE TABLE temperature_translation (
  temp_id       INT NOT NULL,
  lang          VARCHAR(8) NOT NULL,
  name          VARCHAR(50) NOT NULL,
  symbol        VARCHAR(1) NOT NULL,

  FOREIGN KEY (temp_id) 
    REFERENCES temperature(id)
    ON DELETE CASCADE,
  ADD CONSTRAINT UC_Lang UNIQUE (temp_id, lang)
)
